import { FormEvent, useEffect, useState } from "react";
import { AlertCircle, BarChart3, FileText, LoaderCircle, LogOut, Map, Menu, Plus, Sparkles, X } from "lucide-react";
import { ApiError, Category, createIncident, getIncidentSuggestions, getMe, Incident, IncidentInput, IntelligenceResponse, listIncidents, login, logout, register, User } from "./lib/api";
import { Analytics } from "./components/Analytics";
import { AIInsights } from "./components/AIInsights";
import { CityMap } from "./components/CityMap";
import { Evidence } from "./components/Evidence";
import { ServiceStatus } from "./components/ServiceStatus";
import "./styles.css";

const categories = [{ value: "pothole", label: "Pothole" }, { value: "garbage", label: "Garbage" }, { value: "water_supply", label: "Water supply" }, { value: "streetlight", label: "Streetlight" }, { value: "drainage", label: "Drainage" }, { value: "other", label: "Other" }] as const;
const label = (value: string) => value.replace(/_/g, " ").replace(/\b\w/g, letter => letter.toUpperCase());
const message = (error: unknown) => error instanceof ApiError ? error.message : error instanceof Error ? error.message : "Something went wrong. Please try again.";
type View = "reports" | "new" | "analytics" | "map" | "ai";

function AuthScreen({ onAuthenticated }: { onAuthenticated: (user: User) => void }) {
  const [mode, setMode] = useState<"login" | "register">("login"); const [name, setName] = useState(""); const [email, setEmail] = useState(""); const [password, setPassword] = useState(""); const [error, setError] = useState(""); const [busy, setBusy] = useState(false);
  async function submit(event: FormEvent) { event.preventDefault(); if (!email.trim() || !/\S+@\S+\.\S+/.test(email)) return setError("Enter a valid email address."); if (password.length < 8) return setError("Password must be at least 8 characters."); setBusy(true); try { onAuthenticated(mode === "register" ? await register({ name, email, password }) : await login({ email, password })); } catch (reason) { setError(message(reason)); } finally { setBusy(false); } }
  return <main className="auth-page"><section className="auth-card"><div className="brand large"><span className="brand-mark">C</span><span>Civic<span>Lens</span><small>PUBLIC INTELLIGENCE</small></span></div><p className="eyebrow teal">Community reporting</p><h1>{mode === "login" ? "Welcome back" : "Create your account"}</h1><form onSubmit={submit}>{mode === "register" && <label>Full name<input aria-label="Full name" value={name} onChange={event => setName(event.target.value)} /></label>}<label>Email address<input aria-label="Email address" value={email} onChange={event => setEmail(event.target.value)} /></label><label>Password<input aria-label="Password" type="password" value={password} onChange={event => setPassword(event.target.value)} /></label>{error && <div className="alert error" role="alert"><AlertCircle size={17} />{error}</div>}<button className="primary-button" disabled={busy}>{busy && <LoaderCircle className="spin" />}{mode === "login" ? "Sign in" : "Create account"}</button></form><button className="text-button" onClick={() => { setMode(mode === "login" ? "register" : "login"); setError(""); }}>{mode === "login" ? "Need an account? Register" : "Already have an account? Sign in"}</button></section></main>;
}

function IntelligenceSuggestions({ form }: { form: IncidentInput }) {
  const [result, setResult] = useState<IntelligenceResponse | null>(null); const [unavailable, setUnavailable] = useState(false);
  useEffect(() => { if (!form.title || !form.description || !form.address) return; const controller = new AbortController(); setUnavailable(false); getIncidentSuggestions({ title: form.title, description: form.description, category: form.category, address: form.address }, controller.signal).then(setResult).catch(reason => { if (reason.name !== "AbortError") setUnavailable(true); }); return () => controller.abort(); }, [form.title, form.description, form.address, form.category]);
  if (unavailable) return <div className="suggestions-card">Suggestions are temporarily unavailable. You can still submit your report normally.</div>; if (!result) return null;
  return <section className="suggestions-card"><strong>Community intelligence suggestions</strong>{result.predictedCategory && <><strong>Suggested category</strong><p>{label(result.predictedCategory)}</p>{result.categoryConfidence !== undefined && <p>{Math.round(result.categoryConfidence * 100)}% heuristic score</p>}<p>This is a model-generated keyword suggestion, not a confirmed category. Your selected category will not change automatically.</p></>}{(result.duplicateReports ?? []).length > 0 && <><p>Potential duplicate suggestion</p><p><strong>{result.duplicateReports![0].title}</strong> — {Math.round(result.duplicateReports![0].similarityScore * 100)}% similar</p><p>No reports are merged automatically.</p></>}</section>;
}

function ReportForm({ onSaved }: { onSaved: (incident: Incident) => void }) {
  const [form, setForm] = useState<IncidentInput>({ title: "", description: "", category: "other", address: "" }); const [error, setError] = useState("");
  async function submit(event: FormEvent) { event.preventDefault(); setError(""); if (!form.title || !form.description || !form.address) return setError("Complete the title, description, and location."); try { onSaved(await createIncident(form)); } catch (reason) { setError(message(reason)); } }
  const set = (key: keyof IncidentInput, value: string) => setForm(current => ({ ...current, [key]: value }));
  return <form className="incident-form" onSubmit={submit}><label>Title<input aria-label="Title" value={form.title} onChange={event => set("title", event.target.value)} /></label><label>Description<textarea aria-label="Description" value={form.description} onChange={event => set("description", event.target.value)} /></label><label>Category<select aria-label="Category" value={form.category} onChange={event => set("category", event.target.value as Category)}>{categories.map(category => <option key={category.value} value={category.value}>{category.label}</option>)}</select></label><label>Address<input aria-label="Address" value={form.address} onChange={event => set("address", event.target.value)} /></label><IntelligenceSuggestions form={form} />{error && <div className="alert error" role="alert">{error}</div>}<button className="primary-button">Submit report</button></form>;
}

function IntelligencePanel({ incident }: { incident: Incident }) { const intelligence = incident.intelligence; if (!intelligence) return <div className="intelligence-panel muted">AI suggestions were unavailable when this report was submitted.</div>; return <section className="intelligence-panel"><h3>Civic intelligence</h3>{intelligence.predictedCategory && <p><strong>Predicted category:</strong> {label(intelligence.predictedCategory)} {intelligence.categoryConfidence !== undefined && <span>({Math.round(intelligence.categoryConfidence * 100)}% heuristic score)</span>}<small>Model-generated keyword suggestion; the reported category remains authoritative.</small></p>}{intelligence.duplicateReports?.length ? <div><strong>Potential duplicate reports</strong><ul>{intelligence.duplicateReports.map(match => <li key={match.id}><span>{match.title}</span><small>{Math.round(match.similarityScore * 100)}% similar · {match.address}</small></li>)}</ul><small>Similarity is a suggestion, not proof. Reports are never merged automatically.</small></div> : <p className="muted">No potential duplicate was returned for this report.</p>}</section>; }
function IncidentCard({ incident, onSelect }: { incident: Incident; onSelect: () => void }) { return <article className="report-card"><button className="report-card-button" onClick={onSelect}><div><p className="eyebrow teal">{label(incident.category)} · {label(incident.status)}</p><h3>{incident.title}</h3><p>{incident.description}</p><small>{incident.address}</small></div><span aria-hidden="true">›</span></button>{incident.intelligence?.predictedCategory && <p className="report-ai-badge">AI suggestion: {label(incident.intelligence.predictedCategory)}{incident.intelligence.categoryConfidence !== undefined ? ` · ${Math.round(incident.intelligence.categoryConfidence * 100)}%` : ""}</p>}</article>; }

const navItems: Array<{ view: View; label: string; icon: typeof FileText }> = [{ view: "reports", label: "My Reports", icon: FileText }, { view: "map", label: "City Map", icon: Map }, { view: "analytics", label: "Analytics", icon: BarChart3 }, { view: "ai", label: "AI Insights", icon: Sparkles }, { view: "new", label: "Report an Issue", icon: Plus }];

function TopNavigation({ view, open, onNavigate, onToggle, user, onLogout }: { view: View; open: boolean; onNavigate: (view: View) => void; onToggle: () => void; user: User; onLogout: () => void }) { const signOut = async () => { await logout().catch(() => undefined); onLogout(); }; return <header className="top-navigation"><div className="top-navigation-main"><div className="brand top-brand"><span className="brand-mark">C</span><span>Civic<span>Lens</span><small>PUBLIC INTELLIGENCE</small></span></div><nav className={`top-nav-links${open ? " open" : ""}`} aria-label="Primary navigation">{navItems.map(({ view: itemView, label: itemLabel, icon: Icon }) => <button key={itemView} className={view === itemView ? "active" : ""} aria-current={view === itemView ? "page" : undefined} onClick={() => onNavigate(itemView)}><Icon size={17} />{itemLabel}</button>)}</nav><div className="top-account"><span className="user-name">{user.name || user.email}</span><button className="secondary-button" onClick={signOut}><LogOut size={16} />Sign out</button></div><button className="mobile-menu-button" aria-label={open ? "Close navigation menu" : "Open navigation menu"} aria-expanded={open} onClick={onToggle}>{open ? <X size={22} /> : <Menu size={22} />}</button></div></header>; }

function Dashboard({ user, onLogout }: { user: User; onLogout: () => void }) {
  const [view, setView] = useState<View>("reports"); const [menuOpen, setMenuOpen] = useState(false); const [incidents, setIncidents] = useState<Incident[]>([]); const [selected, setSelected] = useState<Incident | null>(null); const [loading, setLoading] = useState(true); const [error, setError] = useState(""); const [notice, setNotice] = useState("");
  useEffect(() => { listIncidents().then(setIncidents).catch(reason => setError(message(reason))).finally(() => setLoading(false)); }, []);
  const navigate = (next: View) => { setView(next); setSelected(null); setMenuOpen(false); };
  const save = (incident: Incident) => { setIncidents(current => [incident, ...current]); setSelected(null); navigate("reports"); setNotice("Report saved successfully."); };
  const currentTitle = view === "analytics" ? "Analytics" : view === "map" ? "City Map" : view === "ai" ? "AI Insights" : view === "new" ? "Report an issue" : selected ? "Incident details" : "Your community reports";
  return <div className="app-shell"><TopNavigation view={view} open={menuOpen} onNavigate={navigate} onToggle={() => setMenuOpen(current => !current)} user={user} onLogout={onLogout} /><main className="main-content"><header className="page-heading"><div><p className="eyebrow">CivicLens workspace</p><h1>{currentTitle}</h1></div><ServiceStatus /></header>{notice && <div className="alert success" role="status">{notice}</div>}{view === "analytics" ? <Analytics /> : view === "map" ? <CityMap incidents={incidents} loading={loading} error={error} /> : view === "ai" ? <AIInsights incidents={incidents} loading={loading} error={error} /> : view === "new" ? <section className="content-card"><p className="eyebrow teal">New report</p><h2>Tell us what needs attention</h2><ReportForm onSaved={save} /></section> : selected ? <section className="content-card"><button className="text-button" onClick={() => setSelected(null)}>← Back to My reports</button><p className="eyebrow teal">{label(selected.category)} · {label(selected.status)}</p><h2>{selected.title}</h2><p>{selected.description}</p><p className="muted">{selected.address}</p><IntelligencePanel incident={selected} /><Evidence incidentId={selected.id} /></section> : <section className="reports-page"><p className="eyebrow teal">Your reports</p>{loading ? <div className="loading-state" role="status">Loading reports…</div> : error ? <div className="alert error" role="alert">Couldn’t load reports. {error}</div> : incidents.length === 0 ? <div className="empty-state"><h2>No reports yet</h2><p>Report a local issue to see it here.</p><button className="primary-button" onClick={() => navigate("new")}>Report your first issue</button></div> : <div className="report-list">{incidents.map(incident => <IncidentCard key={incident.id} incident={incident} onSelect={() => setSelected(incident)} />)}</div>}</section>}</main></div>;
}

export default function App() { const [user, setUser] = useState<User | null>(null); const [checking, setChecking] = useState(true); useEffect(() => { getMe().then(setUser).catch(() => setUser(null)).finally(() => setChecking(false)); }, []); if (checking) return <main className="loading-page" aria-live="polite"><LoaderCircle className="spin" /><p>Checking your session…</p></main>; return user ? <Dashboard user={user} onLogout={() => setUser(null)} /> : <AuthScreen onAuthenticated={setUser} />; }
